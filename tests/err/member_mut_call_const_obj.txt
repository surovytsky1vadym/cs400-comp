struct Point
{
    i32 mut x
    i32 mut y
    fn d2 := () -> i32
    {
        i32 r{x * x + y * y}
        exit r
    }
    fn shift := (i32 dx) -> i32
    {
        x := x + dx
        exit x
    }
}
struct Circle
{
    Point center
    i32 radius
}
Point q{3, 4}
i32 a{q.d2()}
i32 b{q.shift(1)}
exit b
