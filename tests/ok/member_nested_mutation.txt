struct Point
{
    i32 mut x
    i32 mut y
    fn shift := (i32 dx) -> i32
    {
        x := x + dx
        exit x
    }
}
struct Circle
{
    Point mut center
    i32 radius
    fn move := (i32 dx) -> i32
    {
        exit center.shift(dx)
    }
}
