struct Point
{
    i32 mut x
    i32 mut y

    fn d2 := () -> i32
    {
        i32 r{x * x + y * y}
        exit r
    }

    fn shift := (i32 dx) -> i32
    {
        x := x + dx
        exit x
    }
}
struct Circle
{
    Point center
    i32 radius

    fn area2 := () -> i32
    {
        i32 a{radius * radius * 3}
        exit a
    }
}
Point mut p{3, 4}
i32 d{p.d2()}
i32 nx{p.shift(2)}
Circle c{p, 2}
i32 a{c.area2()}
i32 cd{c.center.d2()}
i32 result{d + nx + a + cd}
exit result
