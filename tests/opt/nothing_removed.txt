struct Point
{
    i32 x
    i32 y
    fn sum := () -> i32
    {
        i32 r{x + y}
        exit r
    }
}
fn scale := (i32 v, i32 k) -> i32
{
    i32 r{v * k}
    exit r
}
Point p{3, 4}
i32 mut i{0}
i32 mut acc{0}
while i != 3
{
    acc := acc + scale(p.sum(), 2)
    i := i + 1
}
fn plus := [acc](i32 d) -> i32
{
    i32 r{acc + d}
    exit r
}
exit plus(1)
